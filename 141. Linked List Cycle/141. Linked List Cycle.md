```python

# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution:
    def hasCycle(self, head: Optional[ListNode]) -> bool:
        
        watched_number = set()
        
        while head:
            if head in watched_number:
                return True
            
            else:
                watched_number.add(head)
                head = head.next

        return False

```
