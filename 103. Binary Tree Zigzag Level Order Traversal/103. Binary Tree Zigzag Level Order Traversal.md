### 書いたコード 1
```python
from collections import defaultdict, deque


# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if root is None:
            return []
        
        same_depth_list = defaultdict(list)
        depth = 0
        node_and_depths = deque([(root, depth)])

        while node_and_depths:
            node, depth = node_and_depths.popleft()
            same_depth_list[depth].append(node.val)

            if node.left is not None:
                node_and_depths.append((node.left, depth + 1))
            if node.right is not None:
                node_and_depths.append((node.right, depth + 1))
        
        for depth, values in same_depth_list.items():
            if depth % 2 == 1:
                values.reverse()
        
        return list(same_depth_list.values())
```

深さが奇数のノード達について、最後に出力する前に逆順にするのではなく、都度逆順にするステップを挟む  
今見ている深さと、次の深さで2つキューを用意  

```python
from collections import deque


# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def zigzagLevelOrder(self, root: TreeNode | None) -> list[list[int]]:
        if root is None:
            return []

        frontier = deque([root])
        is_left_to_right = True
        level_order_values = []

        while frontier:
            next_frontier = deque()
            values_at_this_level = []

            while frontier:
                node = frontier.popleft()
                values_at_this_level.append(node.val)

                if node.left is not None:
                    next_frontier.append(node.left)
                if node.right is not None:
                    next_frontier.append(node.right)

            if not is_left_to_right:
                values_at_this_level = list(reversed(values_at_this_level))

            level_order_values.append(values_at_this_level)
            is_left_to_right = not is_left_to_right
            frontier = next_frontier
            
        return level_order_values
```
これを改良したら、自分が出来なかった2つ目のアイデアを使えるのではないか  
`is_left_to_right`によって、左または右の子ノードのどちらを先にスタックに入れるかを決定する  
こうすれば、多少コードが長くなるが、`reversed`関数を使わずに済む  

### 書いたコード 3
```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def zigzagLevelOrder(self, root: TreeNode | None) -> list[list[int]]:
        if root is None:
            return []

        frontier = [root]
        is_left_to_right = True
        level_order_values = []

        while frontier:
            next_frontier = []
            values_at_this_level = []

            while frontier:
                node = frontier.pop()
                values_at_this_level.append(node.val)

                if is_left_to_right:
                    if node.left is not None:
                        next_frontier.append(node.left)
                    if node.right is not None:
                        next_frontier.append(node.right)
                else:
                    if node.right is not None:
                        next_frontier.append(node.right)
                    if node.left is not None:
                        next_frontier.append(node.left)

            level_order_values.append(values_at_this_level)
            is_left_to_right = not is_left_to_right
            frontier = next_frontier

        return level_order_values
```
